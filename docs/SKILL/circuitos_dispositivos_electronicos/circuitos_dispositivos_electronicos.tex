\section*{Skill: Circuitos y Dispositivos Electrónicos}


\subsection*{Cuándo usar este skill}


Este skill debe activarse cuando el usuario plantee problemas o preguntas sobre:


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item Análisis de circuitos eléctricos y electrónicos (DC, transitorio, AC).
  \item Cálculo de tensiones, corrientes y potencias en circuitos.
  \item Aplicación de leyes de Kirchhoff, métodos de nudos y mallas.
  \item Simplificación de circuitos mediante equivalentes Thévenin y Norton.
  \item Análisis de circuitos con resistencias, condensadores, bobinas, diodos, transistores (BJT, MOSFET) y amplificadores operacionales.
  \item Modelado de dispositivos electrónicos (modelos ideales, de pequeña señal, etc.).
  \item Interpretación de hojas de especificaciones de componentes.
\end{itemize}


\subsection*{Conceptos clave}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Tensión y Corriente:} Definiciones, analogías, convenios de signos.
  \item \textbf{Potencia:} Cálculo, convenio de signos (elemento pasivo vs. generador).
  \item \textbf{Componentes ideales:} Resistencia, condensador, bobina, fuentes independientes y dependientes, transformador ideal.
  \item \textbf{Leyes de Kirchhoff:} KCL (nudos) y KVL (mallas).
  \item \textbf{Métodos de análisis:} Nudos, mallas, superposición, Thévenin, Norton.
  \item \textbf{Señales:} Escalón, pulso, rampa, exponencial, sinusoidal; valores medio y eficaz.
  \item \textbf{Dispositivos semiconductores:} Diodo (rectificador, zener), transistor bipolar (BJT), transistor de efecto de campo (MOSFET).
  \item \textbf{Amplificador Operacional:} Modelo ideal, realimentación negativa/positiva, configuraciones básicas (inversor, no inversor, sumador, diferencial).
  \item \textbf{Respuesta transitoria:} Circuitos RC y RL de primer orden, constante de tiempo.
  \item \textbf{Análisis en pequeña señal:} Modelos de pequeña señal del BJT y MOSFET, ganancia, resistencias de entrada/salida.
\end{itemize}


\subsection*{Procedimiento básico de análisis}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Identificar} el tipo de circuito y los componentes involucrados.
  \item \textbf{Determinar} el régimen de operación (DC, transitorio, pequeña señal).
  \item \textbf{Sustituir} los dispositivos por sus modelos adecuados para el régimen de análisis.
  \item \textbf{Aplicar} las leyes de Kirchhoff y los métodos de análisis de circuitos (nudos, mallas, superposición, equivalentes).
  \item \textbf{Plantear y resolver} las ecuaciones resultantes.
  \item \textbf{Verificar} las hipótesis de operación (ej. diodo en directa, transistor en activa).
  \item \textbf{Interpretar} los resultados en el contexto del problema.
\end{enumerate}

\section*{Fórmulas de Circuitos y Dispositivos Electrónicos}


\subsection*{1. Tensión, Corriente y Potencia}


\subsubsection*{Corriente Eléctrica}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $i = \frac{dq}{dt}$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos
q, t = sp.symbols('q t', real=True)
i = sp.Function('i')(t)

# Corriente como derivada de la carga
i_expr = sp.Derivative(q, t)
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} Definición general para corriente instantánea.
\end{itemize}


\subsubsection*{Potencia Eléctrica}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $p = i \cdot v$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos
i, v, t = sp.symbols('i v t', real=True)
p = sp.Function('p')(t)

# Potencia instantánea
p_expr = i * v
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} General para cualquier elemento de dos terminales.
\end{itemize}


\subsection*{2. Ley de Ohm y Resistencia}


\subsubsection*{Ley de Ohm}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $v = i \cdot R$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos
v, i, R = sp.symbols('v i R', real=True)

# Ley de Ohm
ecuacion_ohm = sp.Eq(v, i * R)
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} Solo para resistencias lineales ideales (característica i-v lineal que pasa por el origen).
\end{itemize}


\subsubsection*{Potencia disipada en una resistencia (Efecto Joule)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $P_R = i^2 R = \frac{v^2}{R}$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos
P_R, i, v, R = sp.symbols('P_R i v R', real=True, positive=True)

# Potencia disipada
P_R_expr1 = i**2 * R
P_R_expr2 = v**2 / R
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} Resistencia lineal ideal. La potencia siempre es positiva (disipada).
\end{itemize}


\subsection*{3. Leyes de Kirchhoff}


\subsubsection*{Ley de Corrientes de Kirchhoff (KCL)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $\sum_{k=1}^{n} i_k = 0$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos para n corrientes
n = 3
i_sym = sp.symbols('i0:%d' % n, real=True)

# KCL en un nudo
kcl_expr = sp.Eq(sum(i_sym), 0)
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} Aplicable en cualquier nudo de un circuito de parámetros concentrados.
\end{itemize}


\subsubsection*{Ley de Tensiones de Kirchhoff (KVL)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $\sum_{k=1}^{n} v_k = 0$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos para n tensiones
n = 3
v_sym = sp.symbols('v0:%d' % n, real=True)

# KVL en una malla
kvl_expr = sp.Eq(sum(v_sym), 0)
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Validez:} Aplicable en cualquier malla cerrada de un circuito de parámetros concentrados.
\end{itemize}


\subsection*{4. Asociación de Resistencias}


\subsubsection*{Resistencias en Serie}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Notación:} $R_{eq} = R_1 + R_2 + \dots + R_n$
  \item \textbf{Representación formal:}
\end{itemize}


\begin{lstlisting}
import sympy as sp

# Definir símbolos
R = sp.symbols('R0:3', real=True, positive=True)

# Resistencia equivalente serie
R_eq_serie = sum(R)
\end{lstlisting}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]

\section*{glosario}

\section*{Glosario}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Amplificador Operacional (A.O.):} Circuito integrado activo con dos entradas (inversora y no inversora) y una salida. Modelo ideal: ganancia infinita, impedancia de entrada infinita, impedancia de salida nula.
  \item \textbf{Ancho de histéresis:} Diferencia entre los umbrales de conmutación superior e inferior en un comparador con histéresis (Schmitt trigger).
  \item \textbf{Bobina (Inductor):} Elemento pasivo que almacena energía en forma de campo magnético. Su relación tensión-corriente es $v = L \frac{di}{dt}$.
  \item \textbf{Capacidad (C):} Constante de proporcionalidad entre carga y tensión en un condensador ($q = C v$). Unidad: faradio (F).
  \item \textbf{Circuito:} Interconexión de componentes para realizar una función.
  \item \textbf{Circuito activo:} Aquel capaz de entregar a una carga más potencia que la recibida de la fuente de señal.
  \item \textbf{Circuito equivalente:} Dos circuitos de dos terminales son equivalentes si su característica $i$-$v$ es idéntica para cualquier carga externa.
  \item \textbf{CMRR (Relación de Rechazo al Modo Común):} Medida de la capacidad de un amplificador diferencial para rechazar señales en modo común. $CMRR = 20 \log(\rho)$.
  \item \textbf{Condensador:} Elemento pasivo que almacena energía en forma de campo eléctrico. Su relación tensión-corriente es $i = C \frac{dv}{dt}$.
  \item \textbf{Conductancia (G):} Inversa de la resistencia ($G = 1/R$). Unidad: siemens (S).
  \item \textbf{Constante de tiempo ($\tau$):} Tiempo característico de carga/descarga de un circuito RC o RL. $\tau = RC$ o $\tau = L/R$.
  \item \textbf{Cortocircuito virtual:} En un A.O. ideal con realimentación negativa, la tensión diferencial de entrada tiende a cero ($v_p \approx v_n$), sin que exista corriente entre las entradas.
  \item \textbf{Corriente eléctrica:} Cantidad de carga que atraviesa la sección de un conductor por unidad de tiempo ($i = dq/dt$). Unidad: amperio (A).
  \item \textbf{Diodo:} Dispositivo semiconductor de dos terminales que permite el paso de corriente en un sentido (directa) y lo bloquea en el opuesto (inversa).
  \item \textbf{Diodo Zener:} Diodo diseñado para operar en la región de ruptura, manteniendo una tensión casi constante ($V_z$) para un amplio rango de corrientes inversas.
  \item \textbf{Efecto Early:} Aumento de la corriente de colector en un BJT al aumentar la tensión colector-emisor, debido a la modulación del ancho de la base.
  \item \textbf{Elemento ideal:} Modelo matemático simple que aproxima el comportamiento de un componente real.
  \item \textbf{Fuente dependiente:} Generador cuya magnitud (tensión o corriente) depende de una tensión o corriente en otra parte del circuito.
  \item \textbf{Fuente independiente:} Generador que mantiene una tensión o corriente fija, independientemente del resto del circuito.
  \item \textbf{Ley de Ohm:} Relación lineal entre tensión y corriente en una resistencia ideal ($v = iR$).
  \item \textbf{Ley de corrientes de Kirchhoff (KCL):} La suma algebraica de corrientes en un nudo es cero.
  \item \textbf{Ley de tensiones de Kirchhoff (KVL):} La suma algebraica de tensiones en una malla es cero.
  \item \textbf{Linealidad:} Propiedad de una función que cumple $f(k_1 u_1 + k_2 u_2) = k_1 f(u_1) + k_2 f(u_2)$.
  \item \textbf{Masa:} Punto de referencia del circuito (potencial 0 V).
  \item \textbf{Modelo de pequeña señal:} Circuito lineal incremental que modela el comportamiento de un dispositivo no lineal para pequeñas variaciones alrededor de un punto de operación.
  \item \textbf{Modulación de longitud de canal:} Efecto en MOSFETs donde la corriente de drenador en saturación aumenta ligeramente con $v_{DS}$.
  \item \textbf{Potencia eléctrica:} Energía entregada por unidad de tiempo ($p = i \cdot v$). Unidad: vatio (W).
  \item \textbf{Punto de trabajo (Q):} Valores de corrientes y tensiones en continua de un circuito, en ausencia de señal.
  \item \textbf{Realimentación negativa:} Conexión de la salida a la entrada inversora de un A.O., que tiende a estabilizar la ganancia y mantenerlo en la región lineal.
  \item \textbf{Realimentación positiva:} Conexión de la salida a la entrada no inversora, que tiende a llevar el A.O. a saturación.
  \item \textbf{Resistencia:} Elemento pasivo que disipa energía en forma de calor. Relación $v = iR$. Unidad: ohmio ($\Omega$).
  \item \textbf{Señal:} Magnitud física (tensión o corriente) que contiene información.
  \item \textbf{Superposición:} Principio que establece que la respuesta de un circuito lineal a múltiples fuentes es la suma de las respuestas a cada fuente actuando por separado.
  \item \textbf{Tensión (Voltaje):} Energía por unidad de carga necesaria para mover una carga entre dos puntos. Unidad: voltio (V).
  \item \textbf{Tensión umbral ($V_T$):} En un MOSFET, tensión mínima $v_{GS}$ necesaria para formar el canal conductor. En un diodo ($V_\gamma$), tensión de polarización directa a partir de la cual la corriente se vuelve significativa.
  \item \textbf{Transconductancia ($g_m$):} Parámetro de pequeña señal que relaciona la variación de corriente de salida con la variación de tensión de entrada ($g_m = \Delta i_{out} / \Delta v_{in}$).
  \item \textbf{Transformador:} Dispositivo que transfiere energía entre circuitos mediante acoplamiento magnético. En su modelo ideal, $v_2/v_1 = n$ e $i_2/i_1 = -1/n$.
  \item \textbf{Transistor Bipolar (BJT):} Dispositivo semiconductor de tres terminales (emisor, base, colector) donde la corriente de colector es controlada por la corriente de base.
  \item \textbf{Transistor de Efecto de Campo (MOSFET):} Dispositivo semiconductor de tres terminales (drenador, puerta, surtidor) donde la corriente de drenador es controlada por la tensión de puerta.
  \item \textbf{Valor eficaz (RMS):} Raíz cuadrada del valor medio del cuadrado de una señal periódica. Para una sinusoide, $V_{ef} = A/\sqrt{2}$.
  \item \textbf{Valor medio:} Promedio de una señal periódica en un período.
\end{itemize}


\section*{limites\_aplicabilidad}

\section*{Límites de Aplicabilidad}


\subsection*{1. Ley de Ohm ($v = iR$)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El elemento no es una resistencia lineal ideal (ej. diodo, transistor, resistencia no lineal).
  \item La temperatura del elemento varía significativamente, cambiando su resistividad.
  \item La frecuencia es extremadamente alta, donde los efectos parásitos (inductancia, capacidad) dominan.
  \item \textbf{Supuestos:}
  \item El material es óhmico (relación $v$-$i$ lineal e invariante en el tiempo).
  \item La resistencia es constante.
  \item \textbf{Alternativa:} Usar el modelo no lineal del dispositivo (ej. modelo exponencial del diodo, modelo de Ebers-Moll para BJT).
\end{itemize}


\subsection*{2. Leyes de Kirchhoff (KCL y KVL)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El circuito no es de parámetros concentrados (las dimensiones físicas son comparables a la longitud de onda de la señal).
  \item Existen campos magnéticos variables externos que induzcan tensiones significativas no modeladas.
  \item \textbf{Supuestos:}
  \item La corriente que entra en un nudo es igual a la que sale (no hay acumulación de carga).
  \item La suma de tensiones en una malla es cero (campo eléctrico conservativo).
  \item \textbf{Alternativa:} Usar teoría de líneas de transmisión o análisis de campos electromagnéticos.
\end{itemize}


\subsection*{3. Principio de Superposición}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El circuito contiene elementos no lineales (diodos, transistores fuera de pequeña señal).
  \item Se desea calcular la potencia directamente (la potencia es una función no lineal de la corriente/tensión).
  \item \textbf{Supuestos:}
  \item El circuito es estrictamente lineal (resistencias, condensadores, bobinas, fuentes dependientes lineales).
  \item Las condiciones iniciales en elementos almacenadores de energía son nulas.
  \item \textbf{Alternativa:} Para circuitos no lineales, usar análisis por tramos lineales o métodos numéricos.
\end{itemize}


\subsection*{4. Equivalentes Thévenin y Norton}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El circuito no es lineal.
  \item No existen fuentes dependientes y la resistencia equivalente es cero o infinita (se puede representar, pero son casos límite).
  \item \textbf{Supuestos:}
  \item El circuito a simplificar es lineal.
  \item La parte del circuito a reemplazar no está acoplada magnéticamente con la carga.
  \item \textbf{Alternativa:} Para circuitos no lineales, no existe un equivalente Thévenin/Norton lineal único; se debe analizar el circuito completo.
\end{itemize}


\subsection*{5. Modelo Exponencial del Diodo ($i_D = I_s(e^{v_D/\eta V_T} - 1)$)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item La tensión inversa supera la tensión de ruptura ($V_z$).
  \item La corriente directa es muy alta, donde la resistencia serie parásita ($R_s$) domina.
  \item Se requiere un modelo para análisis de conmutación rápida (se deben incluir capacidades parásitas).
  \item \textbf{Supuestos:}
  \item Unión PN ideal, dopado uniforme, baja inyección.
  \item Temperatura constante y conocida.
  \item \textbf{Alternativa:} Usar modelo con resistencia serie, modelo de ruptura, o modelo dinámico con $C_j$ y $C_s$.
\end{itemize}


\subsection*{6. Modelo de Pequeña Señal del BJT (Híbrido en $\pi$)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item La amplitud de la señal es grande (no se puede linealizar alrededor del punto Q).
  \item El transistor no está polarizado en la región activa.
  \item La frecuencia de la señal es muy alta (se deben considerar las capacidades parásitas $C_\pi$ y $C_\mu$).
  \item \textbf{Supuestos:}
  \item El transistor opera en región activa en continua.
  \item La señal incremental es de pequeña amplitud ($\Delta v_{be} \ll V_T$).
  \item Análisis en baja frecuencia (capacidades parásitas despreciables).
  \item \textbf{Alternativa:} Para gran señal, usar modelos de Ebers-Moll o Gummel-Poon. Para alta frecuencia, usar el modelo híbrido en $\pi$ completo con capacidades.
\end{itemize}


\subsection*{7. Modelo de Pequeña Señal del MOSFET}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item La amplitud de la señal es grande.
  \item El transistor no está polarizado en saturación.
  \item La frecuencia es alta (se deben considerar las capacidades $C_{gs}$, $C_{gd}$, etc.).
  \item \textbf{Supuestos:}
  \item MOSFET operando en saturación en continua.
  \item Señal incremental de pequeña amplitud.
  \item Análisis en baja frecuencia.
  \item \textbf{Alternativa:} Para gran señal, usar las ecuaciones cuadráticas completas. Para alta frecuencia, añadir las capacidades al modelo.
\end{itemize}


\subsection*{8. Amplificador Operacional Ideal (Cortocircuito Virtual)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El A.O. está saturado ($v_o = \pm V_{cc}$).
  \item La realimentación es positiva.
  \item Se analiza el comportamiento en frecuencia (la ganancia $A$ es finita y dependiente de la frecuencia).
  \item \textbf{Supuestos:}
  \item Ganancia diferencial infinita ($A \to \infty$).
  \item Impedancia de entrada infinita ($i_p = i_n = 0$).
  \item Impedancia de salida nula.
  \item Existe realimentación negativa y el A.O. opera en la región lineal.
  \item \textbf{Alternativa:} Usar el modelo de A.O. con ganancia finita, o el modelo de saturaciones si está fuera de la región lineal.
\end{itemize}


\subsection*{9. Fórmula de Carga/Descarga de Circuitos RC ($v_c(t) = V_f + (V_i - V_f)e^{-t/\tau}$)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El circuito no es de primer orden (más de un elemento almacenador de energía).
  \item La excitación no es constante (escalón) o no se puede reducir a un circuito equivalente de primer orden.
  \item La resistencia equivalente vista por el condensador no es constante (ej. circuito con diodos).
  \item \textbf{Supuestos:}
  \item Circuito RC de primer orden.
  \item Fuente de tensión constante aplicada en $t=0$.
  \item Resistencia equivalente $R_{th}$ constante.
  \item \textbf{Alternativa:} Para circuitos de orden superior, resolver la ecuación diferencial completa. Para excitaciones arbitrarias, usar la integral de convolución.
\end{itemize}


\subsection*{10. Fórmula de Carga/Descarga de Circuitos RL ($i_L(t) = I_f + (I_i - I_f)e^{-tR/L}$)}


\begin{itemize}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{NUNCA aplicar si:}
  \item El circuito no es de primer orden.
  \item La excitación no es constante.
  \item La resistencia equivalente vista por la bobina no es constante.
  \item \textbf{Supuestos:}
  \item Circuito RL de primer orden.
  \item Fuente de tensión constante aplicada en $t=0$.
  \item Resistencia equivalente $R_{th}$ constante.
  \item \textbf{Alternativa:} Resolver la ecuación diferencial completa o usar convolución.
\end{itemize}


\section*{metodologias}

\section*{Metodologías de Resolución}


\subsection*{1. Análisis de Circuitos por el Método de Nudos}


\textbf{Aplicación:} Circuitos con resistencias y fuentes de corriente (o fuentes de tensión convertidas).


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Elegir un nudo de referencia} (tierra, 0 V) y asignar tensiones de nudo incógnita ($v_1, v_2, \dots$) a los demás nudos.
  \item \textbf{Aplicar KCL} en cada nudo excepto el de referencia. La suma de corrientes que salen del nudo es cero.
  \item \textbf{Expresar las corrientes} de las resistencias en función de las tensiones de nudo usando la Ley de Ohm ($i = (v_a - v_b)/R$).
  \item Si hay \textbf{fuentes de tensión}, se introduce la corriente de la fuente como incógnita y se añade la ecuación de la tensión de la fuente ($v_a - v_b = v_{fuente}$).
  \item \textbf{Resolver el sistema de ecuaciones} para hallar las tensiones de nudo.
  \item \textbf{Calcular las variables deseadas} (corrientes, tensiones en elementos) a partir de las tensiones de nudo.
\end{enumerate}


\subsection*{2. Análisis de Circuitos por el Método de Mallas}


\textbf{Aplicación:} Circuitos planos con fuentes de tensión (o fuentes de corriente convertidas).


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Asignar corrientes de malla} ($i_1, i_2, \dots$) a cada malla del circuito.
  \item \textbf{Aplicar KVL} en cada malla. La suma de tensiones a lo largo de la malla es cero.
  \item \textbf{Expresar las tensiones} en las resistencias en función de las corrientes de malla usando la Ley de Ohm. Si una resistencia es compartida por dos mallas, la corriente que la atraviesa es la diferencia de las corrientes de malla.
  \item Si hay \textbf{fuentes de corriente}, se introduce la tensión en la fuente como incógnita y se añade la ecuación de la corriente de la fuente ($i_a - i_b = i_{fuente}$).
  \item \textbf{Resolver el sistema de ecuaciones} para hallar las corrientes de malla.
  \item \textbf{Calcular las variables deseadas} a partir de las corrientes de malla.
\end{enumerate}


\subsection*{3. Aplicación del Principio de Superposición}


\textbf{Aplicación:} Circuitos lineales con múltiples fuentes independientes.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Anular todas las fuentes independientes excepto una.}
  \item Fuente de tensión: sustituir por un cortocircuito ($v=0$).
  \item Fuente de corriente: sustituir por un circuito abierto ($i=0$).
  \item \textbf{Las fuentes dependientes NO se anulan.}
  \item \textbf{Calcular la salida} (tensión o corriente deseada) producida únicamente por esa fuente activa.
  \item \textbf{Repetir los pasos 1 y 2} para cada fuente independiente del circuito.
  \item \textbf{Sumar algebraicamente} todas las salidas parciales obtenidas para obtener la salida total.
\end{enumerate}


\subsection*{4. Cálculo del Equivalente Thévenin}


\textbf{Aplicación:} Simplificar un circuito lineal de dos terminales visto desde una carga.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Calcular la tensión Thévenin ($v_{th}$):}
  \item Determinar la tensión de circuito abierto entre los terminales A y B ($v_{oc}$).
  \item $v_{th} = v_{oc}$.
  \item \textbf{Calcular la resistencia Thévenin ($R_{th}$):}
  \item \textbf{Anular todas las fuentes independientes} dentro del circuito (cortocircuito para tensión, circuito abierto para corriente).
  \item \textbf{Mantener las fuentes dependientes.}
  \item Conectar una fuente de tensión de prueba $v_x$ entre los terminales A y B.
  \item Calcular la corriente $i_x$ que suministra la fuente de prueba al circuito.
  \item $R_{th} = v_x / i_x$.
  \item \textbf{Circuito equivalente:} Fuente de tensión $v_{th}$ en serie con la resistencia $R_{th}$.
\end{enumerate}


\subsection*{5. Cálculo del Equivalente Norton}


\textbf{Aplicación:} Simplificar un circuito lineal de dos terminales visto desde una carga.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Calcular la corriente Norton ($i_n$):}
  \item Determinar la corriente de cortocircuito entre los terminales A y B ($i_{sc}$).
  \item $i_n = i_{sc}$.
  \item \textbf{Calcular la resistencia Norton ($R_n$):}
  \item $R_n = R_{th}$ (calculada como en el equivalente Thévenin).
  \item \textbf{Circuito equivalente:} Fuente de corriente $i_n$ en paralelo con la resistencia $R_n$.
\end{enumerate}


\subsection*{6. Análisis de Circuitos con Diodos en DC}


\textbf{Aplicación:} Circuitos con diodos rectificadores y fuentes de continua.


\textbf{Procedimiento (Método de aproximación por tramos lineales):}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Sustituir el diodo} por su modelo de tramos lineales (diodo ideal + $V_\gamma$ + $R_s$).
  \item \textbf{Suponer un estado} para cada diodo (ON: conduce, OFF: no conduce).
  \item \textbf{Analizar el circuito lineal resultante} para la hipótesis elegida.
  \item \textbf{Verificar la hipótesis:}
  \item Si se supuso ON, verificar que $i_D > 0$.
  \item Si se supuso OFF, verificar que $v_D < V_\gamma$.
  \item Si la hipótesis no se cumple, \textbf{cambiar el estado} del diodo y repetir desde el paso 3.
  \item Una vez verificada la hipótesis, la solución es válida.
\end{enumerate}


\subsection*{7. Análisis de Circuitos con Transistores BJT en DC}


\textbf{Aplicación:} Calcular el punto de trabajo (Q) de un circuito con BJT.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Suponer que el BJT opera en región activa} ($V_{BE} \approx 0.7\text{ V}$, $i_C = \beta_F i_B$).
  \item \textbf{Plantear la ecuación de la malla de base} para hallar $I_{BQ}$.
  \item \textbf{Calcular $I_{CQ} = \beta_F I_{BQ}$}.
  \item \textbf{Plantear la ecuación de la malla de colector} para hallar $V_{CEQ}$.
  \item \textbf{Verificar la hipótesis de región activa:}
  \item $V_{CEQ} > 0.2\text{ V}$ (o $V_{CEsat}$).
  \item $V_{BC} < 0.7\text{ V}$ (unión colectora en inversa).
  \item Si no se cumple, el transistor está en \textbf{saturación} o \textbf{corte}. Repetir el análisis con el modelo adecuado:
  \item \textbf{Saturación:} $V_{BE} \approx 0.7\text{ V}$, $V_{CE} \approx 0.2\text{ V}$, $i_C < \beta_F i_B$.
  \item \textbf{Corte:} $i_B = 0$, $i_C = 0$.
\end{enumerate}


\subsection*{8. Análisis de Circuitos con Transistores MOSFET en DC}


\textbf{Aplicación:} Calcular el punto de trabajo (Q) de un circuito con MOSFET.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Suponer que el MOSFET opera en región de saturación} ($i_D = \frac{K}{2}\frac{W}{L}(v_{GS} - V_T)^2$).
  \item \textbf{Plantear las ecuaciones} del circuito para hallar $V_{GSQ}$ e $I_{DQ}$.
  \item \textbf{Calcular $V_{DSQ}$} a partir de la malla de drenador.
  \item \textbf{Verificar la hipótesis de saturación:}
  \item $V_{DSQ} \ge V_{GSQ} - V_T$.
  \item Si no se cumple, el MOSFET está en \textbf{región óhmica}. Repetir el análisis con la ecuación de región óhmica.
\end{enumerate}


\subsection*{9. Análisis de Circuitos con Amplificadores Operacionales Ideales}


\textbf{Aplicación:} Circuitos con A.O. en región lineal (realimentación negativa).


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Verificar} que la realimentación es negativa.
  \item \textbf{Aplicar las reglas del A.O. ideal:}
  \item $v_p = v_n$ (cortocircuito virtual).
  \item $i_p = i_n = 0$ (corrientes de entrada nulas).
  \item \textbf{Plantear KCL} en los nudos de entrada ($v_n$ y $v_p$).
  \item \textbf{Resolver} para encontrar la relación $v_o / v_i$ u otras variables de interés.
  \item \textbf{Verificar} que la salida no excede las tensiones de saturación ($|v_o| < V_{cc}$).
\end{enumerate}


\subsection*{10. Análisis de Respuesta Transitoria en Circuitos RC/RL de Primer Orden}


\textbf{Aplicación:} Circuitos con un solo elemento almacenador de energía (C o L) y fuentes constantes o conmutadas.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Determinar la condición inicial} en $t=0^-$ (justo antes de la conmutación).
  \item Para $C$: $v_C(0^-)$.
  \item Para $L$: $i_L(0^-)$.
  \item \textbf{Determinar la condición inicial en $t=0^+$} (justo después de la conmutación).
  \item $v_C(0^+) = v_C(0^-)$ (continuidad de tensión).
  \item $i_L(0^+) = i_L(0^-)$ (continuidad de corriente).
  \item \textbf{Calcular el valor final} en régimen permanente ($t \to \infty$).
  \item Sustituir $C$ por circuito abierto y $L$ por cortocircuito.
  \item \textbf{Calcular la constante de tiempo $\tau$:}
  \item $\tau = R_{th} C$ o $\tau = L / R_{th}$, donde $R_{th}$ es la resistencia equivalente vista desde el elemento almacenador.
  \item \textbf{Escribir la solución completa:}
  \item $x(t) = x(\infty) + [x(0^+) - x(\infty)] e^{-t/\tau}$
  \item donde $x(t)$ es $v_C(t)$ o $i_L(t)$.
\end{enumerate}


\subsection*{11. Análisis en Pequeña Señal de Amplificadores con BJT}


\textbf{Aplicación:} Calcular ganancia, resistencias de entrada y salida de amplificadores BJT.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Calcular el punto de trabajo en continua (Q)} del transistor (ver Metodología 7).
  \item \textbf{Obtener los parámetros de pequeña señal} en el punto Q:
  \item $g_m = I_{CQ} / V_T$
  \item $r_\pi = \beta_F / g_m$
  \item $r_o = V_A / I_{CQ}$ (si se considera efecto Early)
  \item \textbf{Dibujar el circuito incremental (de pequeña señal):}
  \item Anular fuentes de continua ($V_{CC} \to$ cortocircuito, $I_{CC} \to$ circuito abierto).
  \item Sustituir el BJT por su modelo de pequeña señal (híbrido en $\pi$).
  \item Los condensadores de acoplo y desacoplo se consideran cortocircuitos a la frecuencia de señal.
  \item \textbf{Analizar el circuito incremental} para hallar:
  \item Ganancia de tensión: $A_v = v_o / v_i$.
  \item Resistencia de entrada: $R_i = v_i / i_i$.
  \item Resistencia de salida: $R_o$ (vista desde la carga, anulando la fuente de señal).
\end{enumerate}


\subsection*{12. Análisis en Pequeña Señal de Amplificadores con MOSFET}


\textbf{Aplicación:} Calcular ganancia, resistencias de entrada y salida de amplificadores MOSFET.


\textbf{Procedimiento:}


\begin{enumerate}[leftmargin=*,topsep=2pt,itemsep=1pt]
  \item \textbf{Calcular el punto de trabajo en continua (Q)} del transistor (ver Metodología 8).
  \item \textbf{Obtener los parámetros de pequeña señal} en el punto Q:
  \item $g_m = 2 \sqrt{K \frac{W}{L} I_{DQ}}$
  \item $r_{ds} = 1 / (\lambda I_{DQ})$ (si se considera modulación de canal)
  \item \textbf{Dibujar el circuito incremental:}
  \item Anular fuentes de continua.
  \item Sustituir el MOSFET por su modelo de pequeña señal (fuente de corriente $g_m v_{gs}$ con $r_{ds}$ en paralelo).
  \item Condensadores de acoplo/desacoplo como cortocircuitos.
  \item \textbf{Analizar el circuito incremental} para hallar $A_v$, $R_i$, $R_o$.
\end{enumerate}
