\begin{itemize}[leftmargin=1.5em, itemsep=2pt]

  \item \textbf{R1}: Resistencia de temporización conectada entre Vcc y el pin 7.
  \item \textbf{R2}: Resistencia conectada entre los pines 7 y 6/2.

\end{itemize}
